Aggregatiefuncties

GROUP BY & HAVING

-- Aantal bestellingen per klant
SELECT klant_id, COUNT(*) AS aantal
FROM bestellingen
GROUP BY klant_id;

-- Totaal per categorie
SELECT categorie, SUM(prijs) AS totaal
FROM producten
GROUP BY categorie;

-- HAVING = filter op groep (niet WHERE!)
SELECT klant_id, COUNT(*) AS aantal
FROM bestellingen
GROUP BY klant_id
HAVING COUNT(*) > 5;

JOIN's

-- INNER JOIN: alleen overeenkomende rijen
SELECT k.naam, b.bedrag
FROM klanten k
INNER JOIN bestellingen b ON k.id = b.klant_id;

-- LEFT JOIN: alle klanten, ook zonder bestelling
SELECT k.naam, b.bedrag
FROM klanten k
LEFT JOIN bestellingen b ON k.id = b.klant_id;

-- RIGHT JOIN: alle bestellingen, ook zonder klant
SELECT k.naam, b.bedrag
FROM klanten k
RIGHT JOIN bestellingen b ON k.id = b.klant_id;

-- Meerdere joins
SELECT k.naam, p.naam AS product, b.datum
FROM klanten k
JOIN bestellingen b ON k.id = b.klant_id
JOIN producten p ON b.product_id = p.id;

Subquery's

-- Klanten met bestellingen boven gemiddelde
SELECT naam FROM klanten
WHERE id IN (
    SELECT klant_id FROM bestellingen
    WHERE bedrag > (SELECT AVG(bedrag) FROM bestellingen)
);

-- In FROM (afgeleide tabel)
SELECT gem.stad, gem.gemiddeld_bedrag
FROM (
    SELECT k.stad, AVG(b.bedrag) AS gemiddeld_bedrag
    FROM klanten k
    JOIN bestellingen b ON k.id = b.klant_id
    GROUP BY k.stad
) AS gem
WHERE gem.gemiddeld_bedrag > 50;

Handige tips voor het examen

  • Volgorde: SELECT → FROM → JOIN → WHERE → GROUP BY → HAVING → ORDER BY → LIMIT
  • WHERE werkt op rijen, HAVING werkt op groepen
  • INNER JOIN = alleen matches, LEFT JOIN = alles links
  • Aggregaties (SUM, COUNT etc.) altijd met GROUP BY als je per groep wilt
  • Gebruik aliassen (AS) voor leesbaarheid
  • NULL vergelijk je met IS NULL, nooit met = NULL
