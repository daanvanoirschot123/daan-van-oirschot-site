GROUP BY & HAVING

SELECT klant_id, COUNT(*) AS aantal
FROM bestellingen
GROUP BY klant_id
HAVING COUNT(*) > 5;

JOIN's

SELECT k.naam, b.bedrag
FROM klanten k
INNER JOIN bestellingen b ON k.id = b.klant_id;

SELECT k.naam, b.bedrag
FROM klanten k
LEFT JOIN bestellingen b ON k.id = b.klant_id;

Tabellen & tips

CREATE TABLE studenten (
    id       INT PRIMARY KEY AUTO_INCREMENT,
    naam     VARCHAR(100) NOT NULL,
    email    VARCHAR(150) UNIQUE,
    leeftijd INT
);
ALTER TABLE studenten ADD COLUMN stad VARCHAR(50);
DROP TABLE IF EXISTS studenten;
  • Volgorde: SELECT → FROM → JOIN → WHERE → GROUP BY → HAVING → ORDER BY → LIMIT
  • WHERE op rijen, HAVING op groepen
  • NULL vergelijk je met IS NULL, nooit met = NULL
